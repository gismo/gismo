\section{Executive Summary}
This study reorganizes the G+Smo 2D AS-$G^1$ geometry repository into a strictly classified benchmark suite under \texttt{filedata/domain2d/as\_g1\_benchmark\_suite/} (\textbf{101 unique validated geometries}), presents the mathematical derivation and implementation of four progressive multigrid enhancements (\textbf{Proposals 4, 1, 2, and 3}), and reports systematic benchmarks across V-cycle (\texttt{-c 1}), W-cycle (\texttt{-c 2}), and Multigrid-PCG (\texttt{-s cg}), refinement levels $r \in \{3,4,5,6\}$ (up to $78{,}736$ DOFs), spline degrees $p \in \{3,4,5\}$, and both the first and second biharmonic equations.

\begin{itemize}
  \item \textbf{Near-100\% Direct MG Robustness:} Across all 101 unique domains at $r=3$ ($p=3$, $\nu_1=\nu_2=1$), V-cycle convergence improves from \textbf{74/101} (73.3\%, median 138 iters, 1.408\,s) with Proposal~4 (\texttt{auto}) to \textbf{88/101} (med 33, 0.428\,s) with Proposal~1 (\texttt{ring1}), \textbf{95/101} (med 24, 0.284\,s) with Proposal~3 (\texttt{fd\_ring1}), and \textbf{99/101} (98.0\%, med 17, 0.165\,s) with Proposal~2 (\texttt{line\_ring1}), achieving \textbf{89/89} (100\%) on all simply-connected classes and \textbf{101/101} (med 8, max 52, 0.080\,s) under PCG.
  \item \textbf{True 3-Level W-Cycle Advantage ($r=4$, 61 domains):} On 3-level hierarchies, W-cycle \texttt{line\_ring1} improves median iterations from $24$ (V-cycle) to \textbf{18} and lowers median solve time from $0.551\,$s to \textbf{0.447\,s} (60/61 pass; only the boundary-ill-conditioned \texttt{asg1\_circle\_5patch.xml} stalls for every solver at $r=4$).
  \item \textbf{$h$- and $p$-Robustness:} Under \texttt{mg-W line\_ring1}, iterations stay nearly constant across $r=3\to 6$ (\texttt{4\_patch\_3\_valence}: $17\to 18\to 19\to 20$ up to $64{,}002$ DOFs; \texttt{as\_g1\_pentagon\_p3\_r1}: $18\to 19\to 20$ up to $78{,}736$ DOFs). Across $p=3\to 5$, pointwise \texttt{ring1} deteriorates sharply while \texttt{line\_ring1} and \texttt{fd\_ring1} remain $p$-robust.
  \item \textbf{Second Biharmonic Parity:} Across all 101 domains at $r=3$, \texttt{mg-V line\_ring1} reaches \textbf{99/101} (med 14 iters, 0.241\,s vs.\ 80/101, med 85, 1.563\,s for \texttt{auto}) and \texttt{cg-V line\_ring1} reaches \textbf{101/101} (med 8, max 40, 0.093\,s).
\end{itemize}

\section{Mathematical Motivations, Implementations \& Comparisons of Proposals 1--4}

\subsection{Why Standalone Direct Multigrid Stalls While Multigrid-PCG Succeeds}
In \texttt{gsMultiGrid.hpp}, \texttt{gsMultiGridOp} constructs Galerkin coarse matrices $K_{\ell-1} = (P_{\ell-1}^\ell)^T K_\ell P_{\ell-1}^\ell$ and applies symmetric pre-/post-smoothing (\texttt{step} downward, \texttt{stepT} upward). Consequently, the V-cycle preconditioner $M_{\text{MG}}$ is symmetric positive definite (SPD), and the preconditioned operator $M_{\text{MG}} A$ has real eigenvalues $\lambda_i \in (0, 1]$:
\begin{enumerate}
  \item \textbf{Outlier-Dominated Spectrum of $M_{\text{MG}} A$:} Exact spectral decomposition on the baseline biharmonic operator shows that \textbf{96--99\%} of eigenvalues on 2--4 patch domains (and \textbf{86--87\%} on 6-patch domains) already lie inside the well-conditioned bulk interval $[0.50, 1.00]$ ($\kappa_{\text{bulk}} \le 2$), while only a small cluster of \textbf{4--26 isolated outlier eigenvalues} approaches zero ($\lambda_{\min} \in [0.0088, 0.0697]$; e.g., \texttt{two\_bilinear\_patches} $r=4$: $\lambda_{\min} = 0.04735$, $\kappa = 21.12$; \texttt{square\_6\_patch} $r=3$: $\lambda_{\min} = 0.02387$, $\kappa = 41.90$; \texttt{square\_6\_patch\_non\_bilinear} $r=3$: $\lambda_{\min} = 0.00880$, $\kappa = 113.58$).
  \item \textbf{Memoryless Stationary MG vs.\ Krylov Deflation:} Direct MG is a memoryless fixed-point iteration $e^{(k)} = (I - M_{\text{MG}}A)^k e^{(0)}$ governed permanently by the single worst eigenvalue via $\rho(I - M_{\text{MG}}A) = 1 - \lambda_{\min}$. Reaching relative tolerance $10^{-8}$ requires
  \[
    k_{\text{MG}} \approx \frac{\ln(10^8)}{-\ln(1 - \lambda_{\min})} \approx \frac{18.42}{\lambda_{\min}} = 18.42\,\kappa(M_{\text{MG}}A),
  \]
  predicting $389$ iterations for $\lambda_{\min} = 0.04735$ (matching the baseline's $338$) and $>2{,}000$ iterations for $\lambda_{\min} \le 0.0088$. By contrast, PCG achieves Chebyshev scaling $k_{\text{PCG}} \le \tfrac{1}{2}\ln(2\cdot 10^8)\sqrt{\kappa} \approx 9.56\sqrt{\kappa}$ \emph{and} superlinear Kaniel--Paige--Saad outlier deflation: once CG places polynomial roots near the few small eigenvalues, subsequent steps converge at the bulk factor $(\sqrt{2}-1)/(\sqrt{2}+1) \approx 0.172$.
  \item \textbf{First vs.\ Second Biharmonic Boundary Physics:} In the first biharmonic problem ($u = \partial_n u = 0$ on $\partial\Omega$), clamping $\partial_n u = 0$ eliminates outer-boundary \texttt{EdgeCross} DOFs and forces boundary-adjacent normal curvature onto the coupled \texttt{EdgeTrace}/\texttt{Int(ring0)} interface layer, making baseline direct MG substantially harder than on the second biharmonic problem ($u = \Delta u = 0$, e.g., $338$ vs.\ $89$ iterations on \texttt{two\_bilinear\_patches}).
\end{enumerate}

\subsection{Proposal 4 (\texttt{--smoother auto}): Basis Conditioning, Derivative Scaling \& Symmetric GS}
\paragraph{Mathematical Motivation.}
Before modifying the smoother subspace structure, three algebraic and implementation defects inflated $\kappa(M_{\text{MG}}A)$ in the original drivers and embedding:
\begin{enumerate}
  \item \textbf{Sparsification truncation:} In \texttt{gsAsG1Basis.hpp}, \texttt{collocateCorners} (stride~2 first-biharmonic branch) and \texttt{deriveEdgeEmbedding} called \texttt{.sparseView(1, 1e-4)}, discarding genuine $\mathcal{O}(h)$ and $\mathcal{O}(h^2)$ derivative coefficients on distorted patches (producing $\|T_f P - R T_c\|/\|R T_c\| = 7.83\times 10^{-11}$ on \texttt{4\_patch\_3\_valence\_not\_bilinear}).
  \item \textbf{Extreme diagonal scale disparity ($3.6\times 10^5\times$):} Unscaled physical AS-$G^1$ functionals $(u, \nabla u, \nabla^2 u)$ yield diagonal entries in $K_{\text{free}}$ scaling as $\mathcal{O}(h^{-2})$ for \texttt{Interior}/\texttt{EdgeTrace}, $\mathcal{O}(1)$ for \texttt{EdgeCross}, and spanning $[1.98\times 10^{-2}, 7.17\times 10^3]$ across \texttt{Corner6Jet} on \texttt{square\_6\_patch} ($r=3$), alongside strong off-diagonal $(u_{xx}, u_{yy})$ coupling from $\int (\Delta u)^2$.
  \item \textbf{Half-strength GS and fallback LU coarse solver:} Both drivers passed \texttt{makeGaussSeidelOp} (one forward sweep in \texttt{step}, one backward sweep in \texttt{stepT}) instead of symmetric GS for \texttt{-s mg}, applied a smoother at level $l=0$, and omitted \texttt{setCoarseSolver} (defaulting to \texttt{SparseLU}).
\end{enumerate}

\paragraph{Implementation.}
Tightened all \texttt{.sparseView} drop tolerances in \texttt{gsAsG1Basis.hpp} to \texttt{T(1e-12)}; added uniform per-level parametric cell scaling $h_\ell = \min_p h_{\min}(\mathcal{B}_{\ell,p})$ in \texttt{setupLevelGeometry}/\texttt{setupLevelGeometry2} scaling \texttt{EdgeCross} and first corner derivatives by $h_\ell^{-1}$ and second corner derivatives by $h_\ell^{-2}$ (preserving exact cross-patch matching and intergrid prolongation); set \texttt{mg->setCoarseSolver(makeSparseCholeskySolver(mg->matrix(0)))}; and restricted smoother construction to levels $l \ge 1$ using \texttt{makeSymmetricGaussSeidelOp} for \texttt{-s mg} and \texttt{makeGaussSeidelOp} for \texttt{-s cg}.

\paragraph{Effect.}
Cuts baseline \texttt{mg-V} on \texttt{two\_bilinear\_patches} ($r=4$) from $300\text{ (FAIL)} \to 214$, \texttt{2\_patch\_rectangle} from $103 \to 57$, \texttt{4\_patch\_3\_valence\_not\_bilinear} from $300\text{ (FAIL)} \to 178$, and \texttt{circle\_asg1} from $245$ ($21.5\,$s) $\to 125$ ($11.0\,$s), while improving \texttt{circle\_asg1} PCG solve time by $1.8\times$. Stretched and multi-valence domains (\texttt{square\_6\_patch}, \texttt{weirdo\_multivalence}) still exceed 300 iterations because cross-interface bending locking and anisotropy remain unresolved.

\subsection{Proposal 1 (\texttt{--smoother ring1}): Multiplicative Point GS + Coupled Ring-1 Skeleton}
\paragraph{Mathematical Motivation (Cross-Interface Bending Locking).}
In \texttt{deriveArgyrisBasisEmbedding}, tensor-product B-splines on each patch are partitioned into interior indices $i_1, i_2 \in [2, n_d-3]$ (\texttt{Interior}) and boundary/interface layers $i_n \in \{0,1\}$ and $\{n-2,n-1\}$ spanning \texttt{EdgeTrace} ($\mathcal{V}_{E,0}$), \texttt{EdgeCross} ($\mathcal{V}_{E,1}$), and \texttt{Corner6Jet} ($\mathcal{V}_v$). Because the biharmonic form integrates $(\partial_{nn} u)^2$ and the second normal derivative $\partial_{nn} u$ on interface elements $[0, h]$ couples $B_0, B_1$ directly to the first two interior B-spline layers $B_2, B_3$ (\texttt{Int(ring0)}, $i_n \in \{2,3\}$) with $\mathcal{O}(h^{-4})$ stiffness, relaxing interface DOFs while holding \texttt{Int(ring0)} fixed introduces large artificial normal-curvature jumps.

Mode-localization measurements directly prove this mechanism:
\begin{itemize}
  \item On \texttt{two\_bilinear\_patches} ($r=4$), adding an exact subspace solve on $(\text{Ifc}+\text{Crn})$ \emph{alone} without \texttt{Int(ring0)} reduces $\kappa(M_{\text{MG}}A)$ only from $21.11 \to 19.79$ ($338 \to 318$ MG iterations).
  \item Expanding the block to include \texttt{Int(ring0)} (\textbf{Coupled Ring-1}) drops $\kappa(M_{\text{MG}}A)$ from $21.11 \to 3.27$ and direct MG iterations from $338 \to 43$ ($42$ with Proposal~4 scaling).
\end{itemize}

\paragraph{Implementation \& Complexity (\texttt{gsAsG1SkeletonSmootherOp}).}
\texttt{collectCoupledRing1Dofs} gathers the $\mathcal{O}(\sqrt{N})$ global free index set
\[
  I_{\text{ring1}} = \{\text{free }\mathcal{V}_{E,0},\,\mathcal{V}_{E,1},\,\mathcal{V}_v\} \;\cup\; \{\text{interior DOFs with } i_n \in \{2,3\} \text{ or } \{n-4,n-3\}\}.
\]
In construction, \texttt{gsAsG1SkeletonSmootherOp} forms the boolean selector $P \in \mathbb{R}^{N \times n_{\text{sub}}}$, pre-extracts $R_{\text{sub}} = P^T A \in \mathbb{R}^{n_{\text{sub}} \times N}$, and Cholesky-factorizes $A_{\text{sub}} = R_{\text{sub}} P$ via \texttt{makeSparseCholeskySolver}. Applying \texttt{relaxSkeleton} evaluates the exact restricted residual $r_{\text{sub}} = P^T b - R_{\text{sub}} x$ in $\mathcal{O}(\sqrt{N})$ time and updates $x \leftarrow x + P A_{\text{sub}}^{-1} r_{\text{sub}}$. Symmetry of the pre/post smoother pair in \texttt{gsMultiGridOp} is preserved strictly via multiplicative reversal:
\[
  \texttt{step}: \quad \text{GS}_{\text{fwd}} \;\to\; \text{Ring1}, \qquad\qquad
  \texttt{stepT}: \quad \text{Ring1} \;\to\; \text{GS}_{\text{bwd}}.
\]

\paragraph{Behavior \& Limitations.}
Eliminates interface/vertex locking across all 34 single-vertex stars (34/34 pass at $r=3$, median 23 iters vs.\ 131 for \texttt{auto}) and raises overall 101-domain \texttt{mg-V} pass rate from 74/101 to 88/101. However, its interior smoother remains pointwise GS, so it still struggles on high-aspect-ratio patches (\texttt{sq6\_nb}), curved ducts (\texttt{02\_serpentine\_s\_duct}), and high spline degrees ($p=5$, e.g.\ \texttt{curved\_ring} $89 \to 130 \to 300^*$).

\subsection{Proposal 2 (\texttt{--smoother line\_ring1}): Alternating 1D Strip Blocks + Coupled Ring-1}
\paragraph{Mathematical Motivation (Patch Anisotropy, Double Knots \& Intra-Patch Curvature).}
After Coupled Ring-1 removes interface locking, spectral outliers relocate into patch interiors for three reasons:
\begin{enumerate}
  \item \textbf{Fourth-power Jacobian anisotropy:} Biharmonic directional stiffness scales with $(|J_u|/|J_v|)^4$, producing $81\times$ directional contrast on \texttt{square\_6\_patch} ($|J_u|/|J_v| = 3.0$), $705\times$ on \texttt{square\_6\_patch\_non\_bilinear} ($5.154$), and $383\times$ on \texttt{weirdo\_multivalence} ($4.425$).
  \item \textbf{$C^1$ knot-multiplicity 2 pairing:} Because \texttt{dbasis.uniformRefine(1, 2)} inserts knots with multiplicity~2, cubic $C^1$ interior splines form tightly coupled $2\times 2$ pairs per knot line that stall scalar GS.
  \item \textbf{Spurious intra-patch metric variation & high-$p$ overlap:} Curved ducts (\texttt{02\_serpentine\_s\_duct}) vary Jacobian aspect ratio and off-diagonal metric $g^{12}$ spatially within one patch, while higher degrees $p \ge 4$ widen B-spline support overlap.
\end{enumerate}

\paragraph{Implementation \& Complexity (\texttt{gsAsG1BlockSmootherOp}).}
\texttt{collectInteriorLineAndRing1Blocks} builds width-2, step-2 interior $u$-strips ($\{l_2, l_2+1\} \times [0, m_1-1]$) and $v$-strips ($[0, m_2-1] \times \{l_1, l_1+1\}$) per patch, followed by the Coupled Ring-1 block $I_{\text{ring1}}$ (so 100\% of free DOFs are covered). To avoid building $\mathcal{O}(\sqrt{N})$ sparse transfer matrices, \texttt{gsAsG1BlockSmootherOp} converts $A$ once into RowMajor (\texttt{m\_matRow}), extracts every diagonal submatrix $A_b = P_b^T A P_b$ in $\mathcal{O}(\text{nnz}(A))$ total setup time using a single reusable \texttt{globToLoc} lookup array, and factorizes each banded 1D block ($k \approx 2 m_d$, bandwidth $\mathcal{O}(p)$) and Ring-1 via \texttt{SimplicialLDLT} ($\mathcal{O}(p^2 N)$ total factorization and application cost). Exact $A$-symmetry of the multiplicative Schwarz sweep is enforced by reversing the block sequence between \texttt{step} and \texttt{stepT}:
\[
  E_{\text{step}} = \Big(\prod_{b=K-1}^{0} (I - P_b A_b^{-1} P_b^T A)\Big), \qquad
  E_{\text{stepT}} = \Big(\prod_{b=0}^{K-1} (I - P_b A_b^{-1} P_b^T A)\Big) = E_{\text{step}}^*.
\]
Because every strip solves the \emph{exact physical Galerkin subblock} $P_b^T A P_b$, no damping parameter is needed regardless of patch curvature or $g^{12}$ skewness.

\subsection{Proposal 3 (\texttt{--smoother fd\_ring1}): Interior Fast Diagonalization + Coupled Ring-1}
\paragraph{Mathematical Motivation ($p$-Robust Tensor Inversion).}
In isogeometric analysis, exact Kronecker Fast Diagonalization (FD) on tensor-product B-splines eliminates $p$-dependence in $\mathcal{O}(m_1 m_2(m_1 + m_2))$ operations per patch without forming multi-dimensional Cholesky factors.

\paragraph{Implementation \& Metric/Power Calibration (\texttt{gsAsG1FastDiagSmootherOp}).}
On each patch interior tensor grid $[2..n_1-3] \times [2..n_2-3]$ (whose basis functions satisfy clamped boundary conditions $B_i = B_i' = 0$ at $u,v \in \{0,1\}$), \texttt{assemble1DInteriorMassAndBiharmonic} integrates exact 1D mass $M_d = [\int B_i B_j]$ and 1D biharmonic $B_d = [\int B_i'' B_j'']$ via $(p+1)$-point Gauss quadrature over unique knot spans, then solves the small 1D generalized eigenproblem
\[
  B_d Q_d = M_d Q_d \Lambda_d^{(4)}, \qquad Q_d^T M_d Q_d = I \quad (d \in \{1, 2\}).
\]
Sampling the geometry Jacobian $J$ on a $3\times 3$ patch grid estimates average physical metric weights $c_1 = \overline{|J|(g^{11})^2}$, $c_2 = \overline{|J|(g^{22})^2}$, and $c_0 = \beta\overline{|J|}$ (where $g = J^T J$). By Cauchy--Schwarz on clamped functions ($\int \phi'\psi' \le \|\phi''\|_{L^2}\|\psi\|_{L^2}$), the 2D patch-interior operator is approximated in $(Q_1 \otimes Q_2)$ coordinates by the diagonal
\[
  D_{a,b} = \Big(\sqrt{c_1 \lambda_a^{(4,1)}} + \sqrt{c_2 \lambda_b^{(4,2)}}\Big)^2 + c_0,
\]
and applied via two small dense matrix multiplications $\hat{A}_p^{-1} r = Q_1 \big((Q_1^T R Q_2) \oslash D\big) Q_2^T$. Because spatial metric variation and cross-derivatives $g^{12}$ on curved patches can cause $\hat{A}_p$ to underestimate $A_{pp}$, 12 power-iteration steps in the $A_{pp}$-inner product estimate $\lambda_{\max}(\hat{A}_p^{-1} A_{pp})$ and calibrate $\omega_p = 0.90/\lambda_{\max}$ so every eigenvalue of $\omega_p \hat{A}_p^{-1} A_{pp}$ lies strictly below $1$. The symmetric smoother applies $\text{GS}_{\text{fwd}} \to \text{FD}_{\text{patches}} \to \text{Ring1}$ in \texttt{step} and its exact reverse in \texttt{stepT}.

\section{Behavioral Comparison Across Domain Classes}
\begin{itemize}
  \item \textbf{Single-Patch Domains (\texttt{01\_single\_patch}):} Isotropic curved patches (\texttt{curved\_quad}) converge in $6\text{--}10$ steps for all smoothers, whereas curved anisotropic ducts (\texttt{02\_serpentine\_s\_duct}, AR $4.50$) fail under \texttt{auto}, \texttt{ring1}, and \texttt{fd\_ring1} in MG while \texttt{line\_ring1} converges cleanly in $21\text{--}36$ MG ($11\text{--}15$ CG) steps.
  \item \textbf{Multi-Patch Trees (\texttt{02\_multipatch\_trees}):} On rectilinear or gently curved trees (\texttt{2\_patch\_rectangle}, \texttt{ldomain}, \texttt{two\_bilinear}), \texttt{fd\_ring1} matches or beats \texttt{line\_ring1} because constant-metric FD closely models each patch; on distorted trees (\texttt{comb}, \texttt{curved\_comb}, \texttt{staircase}), \texttt{line\_ring1} provides full $22/22$ robustness.
  \item \textbf{Single-Vertex Stars (\texttt{03\_single\_vertex\_stars}):} Pointwise \texttt{auto} degrades severely as $h\to 0$ or $p$ rises. Coupled Ring-1 eliminates vertex/interface locking ($34/34$), while adding line strips (\texttt{line\_ring1}) preserves $p$-robustness through $d=5$ and achieves nearly $h$-independent W-cycle counts ($18\to 19\to 20$ up to $78{,}736$ DOFs on \texttt{as\_g1\_pentagon\_p3\_r1}).
  \item \textbf{Multi-Vertex Meshes (\texttt{04\_multi\_vertex\_meshes}):} Combining multi-valence 6-jet vertices and high aspect ratio (\texttt{sq6\_nb} AR $5.06$, \texttt{weirdo}, \texttt{filleted\_u\_bracket}) stalls \texttt{auto} ($9/23$) and \texttt{ring1} ($16/23$); \texttt{line\_ring1} achieves $23/23$ (100\%) at $r=3$ and cuts \texttt{circle\_asg1} ($44{,}936$ DOFs at $r=4$) MG solve time from $11.0\,$s to $0.82\,$s.
  \item \textbf{Multiply-Connected Domains (\texttt{05\_multiply\_connected\_holes}):} Closed hole corridors induce strong periodic coupling. \texttt{line\_ring1} maintains $p$-robust single-digit MG counts on \texttt{curved\_ring} ($9\text{--}12$ vs.\ $89\text{--}300^*$ for \texttt{ring1}) and achieves 12/12 CG convergence ($\le 52$ iters) even on 5-hole plates.
\end{itemize}
