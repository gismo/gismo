\begin{abstract}
We present the mathematical formulation, boundary- and corner-DOF treatment, G+Smo implementation, and comprehensive numerical validation of multi-patch direct and multigrid solvers for the \emph{second biharmonic problem} (Navier / simply supported boundary conditions, $u = g_1$ and $\Delta u = g_2$ on $\partial\Omega$) over Analysis-Suitable $G^1$ (AS-$G^1$) multi-patch domains. Unlike the first biharmonic problem, only the trace of the solution is constrained strongly on $\partial\Omega$, while the normal derivative remains free everywhere except at sharp domain corners (Case~A). At smooth boundary vertices connecting two patches (Case~B), the tangential second derivative constraint incorporates the signed boundary curvature so that $u = g_1$ is respected to high order on curved boundaries. Nested Galerkin coarse-grid operators, intergrid transfer matrices, and a symmetric Gauss--Seidel smoother yield a scalable geometric--algebraic multigrid preconditioner. Numerical experiments verify optimal $\mathcal{O}(h^{p+1})$, $\mathcal{O}(h^p)$, and $\mathcal{O}(h^{p-1})$ convergence in $L^2$, $H^1$, and $H^2$ norms, mesh-independent PCG iterations up to $182{,}408$ DOFs, $\mathcal{O}(\nu^{-1/2})$ smoothing-step scaling, zero regression on the original first-biharmonic solvers, and $100\%$ convergence across all $93$ valid AS-$G^1$ multi-patch domains discovered in \texttt{filedata/domain2d} (including planar, curved, non-convex, re-entrant, high-valence, and perforated geometries).
\end{abstract}

\section{Mathematical Formulation}
%======================================================================

\subsection{Strong and Weak Formulations}
Let $\Omega \subset \R^2$ be a bounded Lipschitz domain partitioned into $K$ conforming quadrilateral patches, $\overline{\Omega} = \bigcup_{k=1}^K \overline{\Omega}^{(k)}$, with sufficiently smooth boundary segments and reaction parameter $\beta \ge 0$. We consider the \emph{second biharmonic boundary value problem} (also known as the Navier or simply supported plate problem):
\begin{equation}
\label{eq:strong}
\begin{aligned}
    \beta u + \Delta^2 u &= f && \text{in } \Omega,\\
    u &= g_1 && \text{on } \partial\Omega,\\
    \Delta u &= g_2 && \text{on } \partial\Omega.
\end{aligned}
\end{equation}

Multiplying the PDE by a test function $v \in V_0 := H^2(\Omega) \cap H_0^1(\Omega)$ and integrating by parts twice yields
\begin{equation}
\int_\Omega \Delta^2 u \, v \dd\bx
= \int_\Omega \Delta u \, \Delta v \dd\bx
  + \int_{\partial\Omega} \bigl( (\partial_{\bn} \Delta u) v - \Delta u \, \partial_{\bn} v \bigr) \dd s.
\end{equation}
Since $v = 0$ on $\partial\Omega$, the first boundary term vanishes identically, while substituting the natural boundary condition $\Delta u = g_2$ on $\partial\Omega$ moves the remaining boundary term to the right-hand side. Setting $V_{g_1} := \{ u \in H^2(\Omega) : u|_{\partial\Omega} = g_1 \}$, the weak formulation reads: find $u \in V_{g_1}$ such that
\begin{equation}
\label{eq:weak}
a(u, v) = \ell(v) \qquad \forall v \in V_0,
\end{equation}
where the symmetric positive-definite bilinear form and linear functional are
\begin{equation}
\label{eq:bilin_rhs}
a(u,v) := \int_\Omega \bigl( \Delta u \, \Delta v + \beta u v \bigr) \dd\bx, \qquad
\ell(v) := \int_\Omega f v \dd\bx + \int_{\partial\Omega} g_2 \, \partial_{\bn} v \dd s.
\end{equation}
Using a conforming AS-$G^1$ isogeometric spline space $V_h \subset C^1(\overline{\Omega}) \cap H^2(\Omega)$ eliminates inter-patch normal-derivative jumps and removes the need for interior penalty or Nitsche stabilization terms. For a manufactured boundary datum $g_1 \neq 0$, we split $u_h = u_0 + u_{\mathrm{bnd}}$ with $u_0 \in V_{h,0} := V_h \cap V_0$ and a lifting $u_{\mathrm{bnd}} \in V_h$ satisfying $u_{\mathrm{bnd}}|_{\partial\Omega} \approx g_1$ (computed via boundary $L^2$-projection of the constrained boundary trace DOFs), leading to the reduced problem for $u_0 \in V_{h,0}$:
\begin{equation}
\label{eq:lifted_weak}
a(u_0, v_h) = \ell(v_h) - a(u_{\mathrm{bnd}}, v_h) \qquad \forall v_h \in V_{h,0}.
\end{equation}

%======================================================================
\section{\texorpdfstring{AS-$G^1$}{AS-G1} Space Decomposition and Boundary/Corner Treatment}
%======================================================================

\subsection{Edge Subspaces}
Following the AS-$G^1$ construction on spline spaces of degree $p \ge 3$ and regularity $r_0 = 1$, the global $C^1(\overline{\Omega})$ space decomposes into patch-interior, interface/boundary edge, and vertex functions:
\begin{equation}
V_h = \mathcal{V}_{\mathcal{I}} \oplus
\bigoplus_{E \in \mathcal{E}} \bigl( \mathcal{V}_{E,0} \oplus \mathcal{V}_{E,1} \bigr)
\oplus \bigoplus_{\bv \in \mathcal{V}} \mathcal{V}_{\bv}.
\end{equation}
Along every boundary edge $E \subset \partial\Omega$:
\begin{itemize}
    \item $\mathcal{V}_{E,0}$ spans the boundary trace $u|_E$; all its DOFs (with indices $1 + 2(s-1)$ for edge parameter index $s$) are **constrained** (eliminated into $T_{\mathrm{bnd}}$ and projected against $g_1$ for inhomogeneous data, or zeroed in homogeneous mode).
    \item $\mathcal{V}_{E,1}$ spans the transversal/normal derivative $\partial_{\bn} u|_E$; unlike the first biharmonic problem (where $\partial_{\bn} u = g_2$ is essential and $\mathcal{V}_{E,1}$ is also eliminated), here all $\mathcal{V}_{E,1}$ DOFs (indices $2 + 2(s-1)$) remain **free** in $T_{\mathrm{free}}$.
\end{itemize}

\subsection{Boundary Vertex Conditions: Sharp Corners vs.\ Smooth Vertices}
Each vertex subspace $\mathcal{V}_{\bv}$ has $6$ DOFs representing the $C^2$-jet at $\bv \in \partial\Omega$:
$\{u, u_x, u_y, u_{xx}, u_{xy}, u_{yy}\}$. Let $\bt_1, \bt_2$ denote the unit tangent vectors of the two boundary edges incident at $\bv$ (oriented outward from $\bv$ into their respective edges), with corresponding outward unit normals $\bn_1, \bn_2$. Given an orthonormal or directional local frame $[\mathbf{d}_1, \mathbf{d}_2] \in \R^{2 \times 2}$ at $\bv$, applying the Jacobian chain rule via the $2 \times 2$ frame matrix $Q = [\mathbf{d}_1, \mathbf{d}_2]$ transforms the gradient and Hessian into directional derivatives:
\begin{equation}
\nabla_{\mathbf{d}} u = Q^T \nabla u, \qquad
H_{\mathbf{d}} u = Q^T (\nabla^2 u) Q.
\end{equation}
In \texttt{gsAsG1Basis.hpp} (\texttt{collocateCorners}), the six local collocation functionals at boundary vertex $\bv_i$ are arranged as:
\begin{equation}
\label{eq:corner_6jet}
\begin{aligned}
j = 0 &: u(\bv_i), &
j = 1 &: \partial_{\mathbf{d}_1} u(\bv_i), &
j = 2 &: \partial_{\mathbf{d}_2} u(\bv_i), \\
j = 3 &: \partial_{\mathbf{d}_1\mathbf{d}_1} u(\bv_i), &
j = 4 &: \partial_{\mathbf{d}_2\mathbf{d}_2} u(\bv_i) + c_\kappa \partial_{\mathbf{d}_1} u(\bv_i), &
j = 5 &: \partial_{\mathbf{d}_1\mathbf{d}_2} u(\bv_i).
\end{aligned}
\end{equation}
The vertex classification criterion uses a threshold $\varepsilon > 0$ (default $\varepsilon = 10^{-4}$) on the normal dot product $\bn_1 \cdot \bn_2$:

\paragraph{Case A: Sharp Corner ($\bn_1 \cdot \bn_2 < 1 - \varepsilon$).}
The boundary tangents $\mathbf{d}_1 = \bt_1$ and $\mathbf{d}_2 = \bt_2$ are linearly independent and span $\R^2$. Because $u = g_1$ holds along \emph{both} incident boundary edges $\Gamma_1$ and $\Gamma_2$, differentiating $u$ along each boundary curve fixes the tangential value, first derivatives, and pure tangential second derivatives at $\bv_i$:
\begin{equation}
u(\bv_i), \quad
\partial_{\bt_1} u(\bv_i), \quad
\partial_{\bt_2} u(\bv_i), \quad
\partial_{\bt_1\bt_1} u(\bv_i), \quad
\partial_{\bt_2\bt_2} u(\bv_i)
\end{equation}
are all determined by $g_1$ (eliminated into $T_{\mathrm{bnd}}$: indices $j \in \{0, 1, 2, 3, 4\}$), while the mixed derivative $j = 5$ ($\partial_{\bt_1\bt_2} u(\bv_i)$) is **left free** in $T_{\mathrm{free}}$. (For nearly cusp corners with $|\det([\bt_1, \bt_2])| < 10^{-10}$, $\mathbf{d}_2$ falls back to $\bn_1$.)

\paragraph{Case B: Smooth Boundary Vertex ($\bn_1 \cdot \bn_2 \ge 1 - \varepsilon$).}
The two boundary edges meet smoothly with common outward unit normal $\bn = \frac{\bn_1 + \bn_2}{\|\bn_1 + \bn_2\|}$ and unit tangent $\bt = (n_y, -n_x)^T$. Setting $\mathbf{d}_1 = \bn$ and $\mathbf{d}_2 = \bt$:
\begin{itemize}
    \item Along the arc-length parameterized boundary curve $\gamma(s) \subset \partial\Omega$ passing through $\bv_i$ ($\gamma' = \bt$), differentiating $u(\gamma(s)) = g_1(\gamma(s))$ yields
    \begin{equation}
    \partial_{\bt} u(\bv_i) = \frac{\dd g_1}{\dd s}, \qquad
    \partial_{\bt\bt} u(\bv_i) + c_\kappa \, \partial_{\bn} u(\bv_i) = \frac{\dd^2 g_1}{\dd s^2},
    \end{equation}
    where $c_\kappa := \gamma''(s) \cdot \bn$ is the signed boundary curvature factor (averaged from both incident boundary curves in physical coordinates, \texttt{cKappa = 0.5*(ck0 + ck1)}).
    \item Consequently, the three functionals $j \in \{0, 2, 4\}$ ($u$, $\partial_{\bt} u$, and $\partial_{\bt\bt} u + c_\kappa \partial_{\bn} u$) are **constrained** into $T_{\mathrm{bnd}}$, while the three normal-involved functionals $j \in \{1, 3, 5\}$ ($\partial_{\bn} u$, $\partial_{\bn\bn} u$, $\partial_{\bn\bt} u$) remain **free** in $T_{\mathrm{free}}$.
\end{itemize}
Including the curvature coupling $c_\kappa$ in $j=4$ is essential on curved smooth boundaries (such as disks, ellipses, and curved rings): without $c_\kappa$, setting $\partial_{\bt\bt} u_0 = 0$ enforces $u_0(\gamma(s)) = 0$ only up to $\mathcal{O}(s^2)$ errors proportional to $\kappa \partial_{\bn} u_0$, which would degrade $L^2$/$H^1$ convergence rates.

\section{Geometric--Algebraic Multigrid Method}
%======================================================================

\subsection{Nested Hierarchy and Intergrid Transfer}
Starting from a coarse mesh refinement level $r_{\min}$ and refining uniformly $L$ times to $r_{\max} = r_{\min} + L$, we obtain a sequence of nested AS-$G^1$ free spaces
\begin{equation}
V_{h,0}^{(0)} \subset V_{h,0}^{(1)} \subset \dots \subset V_{h,0}^{(L)},
\end{equation}
with basis embedding operators $T_\ell = T_{\mathrm{free},\ell} : \R^{N_\ell} \to \R^{\hat{N}_\ell}$ mapping free global DOFs to patchwise tensor-product B-spline DOFs. Let $\widehat{P}_{\ell-1}^\ell$ denote the standard B-spline knot-insertion prolongation operator on the fine patchwise tensor-product grid. Because $V_{h,0}^{(\ell-1)} \subset V_{h,0}^{(\ell)}$, each coarse AS-$G^1$ basis function is exactly representable on level $\ell$, giving a unique global prolongation matrix $P_{\ell-1}^\ell \in \R^{N_\ell \times N_{\ell-1}}$ solving
\begin{equation}
\label{eq:prolongation}
T_\ell P_{\ell-1}^\ell = \widehat{P}_{\ell-1}^\ell T_{\ell-1},
\end{equation}
computed via sparse QR / normal-equation solve on $T_\ell$. Note that level $0 \dots L-1$ construct only the geometry-level AS-$G^1$ bases and prolongation operators $P_{\ell-1}^\ell$, while the PDE bilinear form $K_L$ and load vector $\mathbf{f}_L$ are assembled strictly on the finest grid $L$.

\subsection{Galerkin Coarse Operators, Smoother, and Cycle Convergence}
Coarse stiffness matrices are generated algebraically via the Galerkin variational property:
\begin{equation}
K_{\ell-1} = (P_{\ell-1}^\ell)^T K_\ell P_{\ell-1}^\ell, \qquad \ell = L, L-1, \dots, 1.
\end{equation}
On each level $\ell \ge 1$, the smoother is $\nu$ pre-smoothing steps and $\nu$ post-smoothing steps of **symmetric Gauss--Seidel** (\texttt{makeGaussSeidelOp} with symmetric forward-backward sweep), which preserves symmetry and positive-semidefiniteness of the multigrid preconditioner. On the coarsest grid $\ell = 0$, the system $K_0 \mathbf{e}_0 = \mathbf{r}_0$ is solved directly using sparse Cholesky factorization (\texttt{gsSparseSolver<>::SimplicialLDLT}).

For a fourth-order elliptic operator ($\beta u + \Delta^2 u$) smoothed by Gauss--Seidel, the smoothing property with respect to the $A$-norm satisfies $\|\widetilde{K}_\ell^{-1} S_\ell^\nu\| \lesssim \nu^{-1/2}$ (since the biharmonic operator scales as $h^{-4}$ while single-step point smoothing damps high-frequency modes like $I - c K_\ell / \mathrm{diag}(K_\ell)$), yielding a two-grid / W-cycle contraction factor bounded by $q \lesssim C / \sqrt{\nu}$ independently of mesh size $h$ \cite{SognTakacs}. When wrapped as a preconditioner inside Preconditioned Conjugate Gradients (PCG), the iteration count remains bounded with respect to $h$ and decays proportionally to $\nu^{-1/2}$ as $\nu$ increases.

\section{G+Smo Implementation Summary}
%======================================================================

The implementation spans two shared AS-$G^1$ headers and two new example drivers:
\begin{itemize}
    \item \texttt{src/gsModeling/gsAsG1Domain.hpp}: Added \texttt{gsBoundaryCornerData<T>} and \texttt{getBoundaryCornerData} to classify boundary vertices into Case~A (sharp, \texttt{isSmooth=false}) or Case~B (smooth, \texttt{isSmooth=true}) and compute local frames $(\mathbf{d}_1, \mathbf{d}_2)$ and signed curvature $c_\kappa$. The existing \texttt{getOuterNormalDerivative} routine was left untouched.
    \item \texttt{src/gsModeling/gsAsG1Basis.hpp}:
    \begin{itemize}
        \item Extended \texttt{collocateCorners} while keeping an exact unchanged branch when \texttt{normals.cols() == 8} (guaranteeing zero change to the first biharmonic code path), and added support for general $2\times 2$ local frames with curvature correction (\texttt{normals.cols() == 16} or \texttt{20}).
        \item Extended \texttt{cornerConditionType} with \texttt{valuesSharp} ($5$ constrained, $1$ free), \texttt{valuesSmooth} ($3$ constrained, $3$ free), and \texttt{derivatives} ($2$ constrained).
        \item Extended \texttt{makeMapperForArgyrisBasis} to support boundary condition types \texttt{"Values"} (eliminating edge $\mathcal{V}_{E,0}$ only and keeping $\mathcal{V}_{E,1}$ free) and \texttt{"Derivatives"}.
    \end{itemize}
    \item \texttt{examples/as\_g1\_biharmonic2\_example.cpp}: Multi-patch direct solver (\texttt{SparseLU}) and $h$-refinement error analyzer ($L^2, H^1, H^2$ norms and convergence orders) with CLI options \texttt{-f}, \texttt{-d}, \texttt{-m}, \texttt{-r}, \texttt{-a}, \texttt{-b/--beta}, \texttt{-e/--eps}, \texttt{--homo}, \texttt{--plot}, \texttt{-o}.
    \item \texttt{examples/as\_g1\_biharmonic2\_multigrid\_example.cpp}: Multi-patch geometric--algebraic multigrid solver supporting PCG (\texttt{-s cg}) and standalone MG (\texttt{-s mg}), V/W-cycles (\texttt{-c 1|2}), configurable pre/post smoothing (\texttt{--pre/--post}), reaction parameter \texttt{-b/--beta}, direct Cholesky cross-check (\texttt{--verify}), and benchmark suites (\texttt{--all --suite classic|generator|noholes|holes|all}, \texttt{-t <tsv>}).
\end{itemize}
